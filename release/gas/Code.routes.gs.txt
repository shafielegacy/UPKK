// _handleApi below. GAS Admin UI uses authenticated script.run, not this API.
function doPost(e) {
  const deny=()=>ContentService.createTextOutput(JSON.stringify({
    success:false,message:'API pendaftaran tidak tersedia untuk permintaan ini.'
  })).setMimeType(ContentService.MimeType.JSON);
  try {
    const raw=e&&e.postData&&e.postData.contents;
    if(typeof raw!=='string'||raw.length>4*1024*1024+16384)return deny();
    const body=JSON.parse(raw);
    if(!body||body.action!=='pagesNativeEdaftar')return deny();
    return pagesNativeEdaftarPost_(body);
  }catch(e){return deny();}
}
function _handleApi(action,params) {

  return ContentService.createTextOutput(JSON.stringify({
    success:false,message:'API awam tidak tersedia. Gunakan portal UPKK rasmi.'
  })).setMimeType(ContentService.MimeType.JSON);
}
